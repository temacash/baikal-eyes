function drawFrostPattern(cv) {
  if (!cv) return;
  const w = cv.width = Math.round(innerWidth * DPR), h = cv.height = Math.round(innerHeight * DPR);
  const ctx = cv.getContext('2d'), r = rng(2024);
  const g = ctx.createRadialGradient(w / 2, h * .46, Math.min(w, h) * .12, w / 2, h * .5, Math.max(w, h) * .72);
  g.addColorStop(0, 'rgba(200,235,250,.04)'); g.addColorStop(.55, 'rgba(190,228,245,.15)'); g.addColorStop(1, 'rgba(214,242,252,.42)');
  ctx.fillStyle = g; ctx.fillRect(0, 0, w, h);

  function branch(x, y, ang, len, depth, wid) {              // дендрит: ветвь с боковыми иглами
    if (depth <= 0 || len < 3) return;
    const x2 = x + Math.cos(ang) * len, y2 = y + Math.sin(ang) * len;
    ctx.strokeStyle = 'rgba(228,247,255,' + (.05 + depth * .045) + ')';
    ctx.lineWidth = wid; ctx.lineCap = 'round';
    ctx.beginPath(); ctx.moveTo(x, y); ctx.lineTo(x2, y2); ctx.stroke();
    const n = 2 + ((r() * 2) | 0);
    for (let i = 1; i <= n; i++) {
      const t = i / (n + 1), bx = x + (x2 - x) * t, by = y + (y2 - y) * t, s = len * (.26 + r() * .3);
      for (const k of [-1, 1]) {
        const a = ang + k * (Math.PI / 3) * (.8 + r() * .4);
        const ex = bx + Math.cos(a) * s, ey = by + Math.sin(a) * s;
        ctx.beginPath(); ctx.moveTo(bx, by); ctx.lineTo(ex, ey); ctx.stroke();
        if (depth > 2 && r() > .62) branch(ex, ey, a, s * .55, depth - 2, Math.max(.4, wid * .7));
      }
    }
    branch(x2, y2, ang + (r() - .5) * .5, len * .72, depth - 1, Math.max(.4, wid * .82));
  }

  for (let i = 0; i < 30; i++) {                              // иней нарастает от краёв внутрь
    const side = i % 4, t = r(); let x, y, a;
    if (side === 0) { x = t * w; y = 0; a = Math.PI / 2 + (r() - .5) * .9; }
    else if (side === 1) { x = w; y = t * h; a = Math.PI + (r() - .5) * .9; }
    else if (side === 2) { x = t * w; y = h; a = -Math.PI / 2 + (r() - .5) * .9; }
    else { x = 0; y = t * h; a = (r() - .5) * .9; }
    branch(x, y, a, Math.min(w, h) * (.08 + r() * .13), 4, Math.max(.8, DPR));
  }
  for (let i = 0; i < 170; i++) {                             // кристаллики
    const x = r() * w, y = r() * h, s = (1 + r() * 2.4) * DPR;
    ctx.globalAlpha = .08 + r() * .45; ctx.fillStyle = '#EAF7FD';
    ctx.beginPath();
    for (let k = 0; k < 6; k++) { const a = k * Math.PI / 3; ctx[k ? 'lineTo' : 'moveTo'](x + Math.cos(a) * s, y + Math.sin(a) * s); }
    ctx.closePath(); ctx.fill();
  }
  ctx.globalAlpha = 1;
}

function frostIntro() {
  const fr = $('#frost'), lg = $('#frostlogo'), root = document.documentElement;
  const kill = () => { root.classList.remove('freezing'); if (fr) fr.remove(); if (lg) lg.remove(); };
  if (!fr) return;
  let seen = false;
  try { seen = sessionStorage.getItem('be-frost') === '1'; } catch (e) {}
  if (seen || matchMedia('(prefers-reduced-motion: reduce)').matches) { kill(); return; }
  try { sessionStorage.setItem('be-frost', '1'); } catch (e) {}

  root.classList.add('freezing');
  drawFrostPattern($('#frostice'));
  const set = (k, v) => root.style.setProperty(k, v);
  let progress = 0;

  const tick = t0 => requestAnimationFrame(function run(now) {   // индикатор, пока грузится фото
    if (melting) return;
    progress = Math.min(70, (now - t0) / 900 * 70);
    set('--f-p', progress.toFixed(1) + '%');
    requestAnimationFrame(run);
  });
  let melting = false;
  tick(performance.now());

  const img = document.querySelector('.hero__bg img');
  const loaded = img && !img.complete
    ? new Promise(res => { img.addEventListener('load', res, { once: true }); img.addEventListener('error', res, { once: true }); })
    : Promise.resolve();
  const minWait = new Promise(res => setTimeout(res, 700));
  const timeout = new Promise(res => setTimeout(res, 2800));

  Promise.race([Promise.all([loaded, minWait]), timeout]).then(() => {
    melting = true;
    const t0 = performance.now(), dur = 2000,
      ease = t => (t < .5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2);
    requestAnimationFrame(function step(now) {
      const p = Math.min(1, (now - t0) / dur), e = ease(p);
      const r1 = e * 190, r0 = Math.max(0, r1 - 16);
      set('--f-r0', r0.toFixed(1) + '%');
      set('--f-r1', r1.toFixed(1) + '%');
      set('--f-blur', (26 * (1 - e)).toFixed(1) + 'px');
      set('--f-bg', (.62 * (1 - e)).toFixed(3));
      set('--f-ice', Math.max(0, 1 - e * 1.4).toFixed(3));
      set('--f-p', (progress + (100 - progress) * Math.min(1, p / .45)).toFixed(1) + '%');
      const vis = p < .38 ? 1 : Math.max(0, 1 - (p - .38) / .4);
      set('--f-lg', vis.toFixed(3));
      set('--f-lgs', (1 + (1 - vis) * .07).toFixed(3));
      set('--f-lgb', ((1 - vis) * 8).toFixed(1) + 'px');
      if (p < 1) requestAnimationFrame(step); else kill();
    });
  });
}
