/* ---------- гидратация страницы ---------- */
function hydrate(rt) {
  observe(app); counters(app);
  plx = $$('[data-parallax]', app).map(el => ({ el, f: parseFloat(el.dataset.parallax) }));
  parallax();

  // карточки маршрутов
  $$('.card[data-href]', app).forEach(c => {
    const go = () => { location.hash = c.dataset.href; };
    c.addEventListener('click', go);
    c.addEventListener('keydown', e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); go(); } });
  });

  if (rt.r === 'home') {
    const setMood = k => {
      const m = MOODS.find(x => x.k === k);
      $$('.mood__it', app).forEach(b => { const on = b.dataset.mood === k; b.classList.toggle('on', on); b.setAttribute('aria-selected', on); });
      $$('[data-mood-c]', app).forEach(c => c.classList.toggle('on', c.dataset.moodC === k));
      $('#moodtext').textContent = m.d;
      $('#moodpicks').innerHTML = m.pick.map(s => { const t = bySlug(s); return `${t.title} →`; }).join('');
    };
    $$('.mood__it', app).forEach(b => {
      b.addEventListener('click', () => setMood(b.dataset.mood));
      b.addEventListener('mouseenter', () => setMood(b.dataset.mood));
    });
    setMood(MOODS[0].k);
  }

  if (rt.r === 'tours' || rt.r === 'excursions') {
    const kind = rt.r === 'tours' ? 'tour' : 'excursion';
    const base = rt.r === 'tours' ? '#/tours' : '#/excursions';
    const grid = $('#tourgrid');
    $$('.chip', app).forEach(ch => ch.addEventListener('click', () => {
      const f = ch.dataset.f;
      $$('.chip', app).forEach(c => { c.classList.toggle('on', c === ch); c.setAttribute('aria-selected', c === ch); });
      history.replaceState(null, '', base + (f === 'all' ? '' : '?f=' + f));
      const list = byKind(kind).filter(t => f === 'all' || t.cat.includes(f));
      grid.style.opacity = '0'; grid.style.transform = 'translateY(10px)'; grid.style.transition = 'opacity .3s, transform .3s';
      setTimeout(() => {
        grid.innerHTML = list.map(cardTpl).join('') || '
В этой категории пока нет маршрутов.
';
        grid.style.opacity = '1'; grid.style.transform = 'none';
        observe(grid);
        $$('.card[data-href]', grid).forEach(c => c.addEventListener('click', () => { location.hash = c.dataset.href; }));
      }, 300);
    }));
  }

  if (rt.r === 'tour') {
    const open = $('#bookopen'), fm = $('#bookform');
    if (open) open.addEventListener('click', () => {
      fm.classList.remove('hide');
      fm.scrollIntoView({ behavior: 'smooth', block: 'center' });
      setTimeout(() => $('#bf-name').focus(), 400);
    });
    bindForm($('#tourform'), $('#bf-sent'), $('#tourform') && $('#tourform').dataset.tour);
    // лайтбокс галереи
    $$('.gallery figure', app).forEach(f => {
      const show = () => {
        const lb = $('#lb'), c = $('#lbc');
        const w = Math.min(1280, Math.round(innerWidth * .9)), h = Math.round(w * .62);
        c.style.width = w + 'px'; c.style.height = h + 'px';
        paintScene(c, f.dataset.lb, +f.dataset.seed);
        lb.classList.add('on'); lb.setAttribute('aria-hidden', 'false'); $('#lbx').focus();
      };
      f.addEventListener('click', show);
      f.addEventListener('keydown', e => { if (e.key === 'Enter') show(); });
    });
  }

  if (rt.r === 'packing') {
    $$('.seasontab', app).forEach(b => b.addEventListener('click', () => {
      const k = b.dataset.season;
      $$('.seasontab', app).forEach(x => { x.classList.toggle('on', x === b); x.setAttribute('aria-selected', x === b); });
      $$('[data-season-panel]', app).forEach(p => p.classList.toggle('hide', p.dataset.seasonPanel !== k));
    }));
  }

  if (rt.r === 'contacts') bindForm($('#cform'), $('#cf-sent'));
}

function closeMenu() { $('#mmenu').classList.remove('on'); $('#burger').classList.remove('on'); $('#burger').setAttribute('aria-expanded', 'false'); $('#mmenu').setAttribute('aria-hidden', 'true'); document.body.style.overflow = ''; }
$('#burger').addEventListener('click', () => {
  const on = $('#mmenu').classList.toggle('on');
  $('#burger').classList.toggle('on', on);
  $('#burger').setAttribute('aria-expanded', String(on));
  $('#mmenu').setAttribute('aria-hidden', String(!on));
  document.body.style.overflow = on ? 'hidden' : '';
});
$('#mmenu').addEventListener('click', e => { if (e.target.closest('a')) closeMenu(); });
const closeLb = () => { $('#lb').classList.remove('on'); $('#lb').setAttribute('aria-hidden', 'true'); };
$('#lbx').addEventListener('click', closeLb);
$('#lb').addEventListener('click', e => { if (e.target.id === 'lb') closeLb(); });
document.addEventListener('keydown', e => { if (e.key === 'Escape') { closeLb(); closeMenu(); } });
